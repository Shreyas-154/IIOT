//Arduino 2
String data;
float temp;
int ledPin = 13;
float threshold = 30.0;

void setup() {
  Serial.begin(9600);
  pinMode(ledPin, OUTPUT);
}

void loop() {
  if (Serial.available()) {
    data = Serial.readStringUntil('\n');
    temp = data.toFloat(); // convert string to number
    
    Serial.print("Temperature: ");
    Serial.println(temp);
    
    if (temp > threshold) {
      digitalWrite(ledPin, HIGH);
      Serial.println("ALERT: Temperature High!");
    } else {
      digitalWrite(ledPin, LOW);
    }
  }
}
